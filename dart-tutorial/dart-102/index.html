Class আর ছোট constructor

💡 Dart-এ new লেখা লাগে না: Product(...)।
Dart
class Product {
  final String name;
  final int price;
  int stock;

  Product(this.name, this.price, {this.stock = 0});

  int get stockValue => price * stock;

  void sell(int qty) {
    stock -= qty;
  }
}

void main() {
  final p = Product('Redmi 13', 17500, stock: 5);
  p.sell(2);
  print('${p.name}: স্টক ${p.stock}, মূল্য ${p.stockValue}');
}

Named আর factory constructor

একাধিক উপায়ে object বানাতে named constructor: Product.used(...)। factory চাইলে নতুন object না বানিয়ে অন্য কিছু ফেরত দিতে পারে — যেমন Map থেকে বানানো।

PHP/Laravel-এ যেমন Laravel-এর Product::make([...]) বা static constructor-এর মতো। toString() = PHP-র __toString()।
Dart
class Product {
  final String name;
  final int price;
  final bool used;

  Product(this.name, this.price) : used = false;
  Product.used(this.name, this.price) : used = true;

  factory Product.fromMap(Map<String, dynamic> m) =>
      Product(m['name'] as String, m['price'] as int);

  @override
  String toString() => '$name ($price টাকা${used ? ', used' : ''})';
}

void main() {
  print(Product('Galaxy A15', 21000));
  print(Product.used('iPhone 12', 48000));
  print(Product.fromMap({'name': 'Vivo Y28', 'price': 18000}));
}

Private: নামের আগে _

Dart-এ private keyword নেই — নাম _ দিয়ে শুরু হলে সেটা ঐ file-এর বাইরে দেখা যায় না। Getter/setter দিয়ে নিয়ন্ত্রিত প্রবেশ।

Dart
class Wallet {
  int _balance = 0;

  int get balance => _balance;

  set deposit(int amount) {
    if (amount <= 0) throw ArgumentError('টাকা ০-এর বেশি দিন');
    _balance += amount;
  }
}

void main() {
  final w = Wallet();
  w.deposit = 500;
  w.deposit = 200;
  print('ব্যালেন্স: ${w.balance}');
  try {
    w.deposit = -50;
  } on ArgumentError catch (e) {
    print('আটকাল: ${e.message}');
  }
}

extends আর abstract

PHP/Laravel-এ যেমন PHP-র abstract class আর extends একই। @override শুধু চিহ্ন, ভুল নাম লিখলে Dart সতর্ক করে।
Dart
abstract class Payable {
  int get amount;
  String get label;
  String describe() => '$label: $amount টাকা';
}

class Invoice extends Payable {
  Invoice(this.no, this.amount);
  final int no;
  @override
  final int amount;
  @override
  String get label => 'Invoice #$no';
}

class ServiceBill extends Payable {
  ServiceBill(this.parts, this.labour);
  final int parts, labour;
  @override
  int get amount => parts + labour;
  @override
  String get label => 'Service';
}

void main() {
  final List<Payable> items = [Invoice(101, 23500), ServiceBill(1800, 500)];
  for (final i in items) {
    print(i.describe());
  }
}

Mixin = PHP trait

mixin হলো এমন কোড যা অনেক class-এ with দিয়ে মিশিয়ে দেওয়া যায় — inheritance ছাড়াই। PHP-র trait।

PHP/Laravel-এ যেমন class Customer { use Timestamps, SoftDeletes; } — Laravel model-এর trait-গুলোর মতোই। .. হলো cascade: একই object-এ পরপর method ডাকা।
Dart
mixin Timestamps {
  DateTime? createdAt;
  void touch() => createdAt = DateTime(2026, 10, 21, 10, 30);
}

mixin SoftDeletes {
  bool deleted = false;
  void softDelete() => deleted = true;
}

class Customer with Timestamps, SoftDeletes {
  Customer(this.name);
  final String name;
}

void main() {
  final c = Customer('Karim')..touch()..softDelete();
  print('${c.name}: তৈরি ${c.createdAt}, মোছা? ${c.deleted}');
}
আউটপুট
Karim: তৈরি 2026-10-21 10:30:00.000, মোছা? true

Enhanced enum

Dart enum-এ field আর method থাকতে পারে — PHP 8.1-এর backed enum-এর চেয়েও শক্তিশালী।

Dart
enum PaymentMethod {
  cash('নগদ', 0),
  bkash('bKash', 1.5),
  nagad('Nagad', 1.5),
  card('Card', 2.0);

  const PaymentMethod(this.label, this.feePercent);
  final String label;
  final double feePercent;

  double feeOn(int amount) => amount * feePercent / 100;
}

void main() {
  for (final m in PaymentMethod.values) {
    print('${m.label.padRight(6)} ১০,০০০ টাকায় চার্জ ${m.feeOn(10000)}');
  }
  print(PaymentMethod.values.byName('bkash').label);
}

Extension: পুরনো type-এ নতুন method

নিজের লেখা না এমন class-এও method যোগ করা যায় — যেমন int-এ টাকা লেখার format।

PHP/Laravel-এ যেমন PHP-তে এটা সম্ভব না (built-in type-এ method যোগ)। Laravel-এর Str::macro() কাছাকাছি ধারণা।
Dart
extension Taka on int {
  String get taka {
    final s = toString();
    if (s.length <= 3) return '৳$s';
    final last3 = s.substring(s.length - 3);
    var rest = s.substring(0, s.length - 3);
    final parts = <String>[];
    while (rest.length > 2) {
      parts.insert(0, rest.substring(rest.length - 2));
      rest = rest.substring(0, rest.length - 2);
    }
    if (rest.isNotEmpty) parts.insert(0, rest);
    return '৳${parts.join(',')},$last3';
  }
}

void main() {
  print(170400.taka);
  print(12500000.taka);
  print(650.taka);
}

class Paginated<T> {
  Paginated(this.items, {required this.page, required this.total});

void main() {

Record: কয়েকটা মান একসাথে ফেরত

(int, String) বা নামসহ ({int total, int count}) — ছোট class না বানিয়েই কয়েকটা মান ফেরত।

PHP/Laravel-এ যেমন PHP-তে array ফেরত দিয়ে [$name, $price] = cheapest() করতাম — Dart-এ এটা type-নিরাপদ।
Dart
({int total, int count}) summarize(List<int> sales) =>
    (total: sales.fold(0, (a, b) => a + b), count: sales.length);

(String, int) cheapest(Map<String, int> prices) {
  final e = prices.entries.reduce((a, b) => a.value <= b.value ? a : b);
  return (e.key, e.value);
}

void main() {
  final s = summarize([18150, 2100, 24150]);
  print('মোট ${s.total}, invoice ${s.count}টা');

  final (name, price) = cheapest({'Redmi 13': 17500, 'Nokia 105': 2100, 'Vivo Y28': 18000});
  print('সবচেয়ে সস্তা: $name ($price)');
}

Pattern matching

switch-এ শুধু মান না, আকৃতিও মেলানো যায়: list, map, record, type — আর সাথে when শর্ত।

Dart
String describe(Object? command) => switch (command) {
  ['sell', String item, int qty] when qty > 0 => '$item × $qty বিক্রি',
  ['sell', _, _] => 'পরিমাণ ভুল',
  {'type': 'refund', 'amount': int a} => '$a টাকা ফেরত',
  int n => 'শুধু একটা সংখ্যা: $n',
  null => 'কিছু নেই',
  _ => 'বুঝিনি',
};

void main() {
  print(describe(['sell', 'Charger', 2]));
  print(describe(['sell', 'Charger', 0]));
  print(describe({'type': 'refund', 'amount': 650}));
  print(describe(42));
  print(describe(null));
  print(describe('hello'));
}

Exception: throw, try, catch

PHP/Laravel-এ যেমন try { } catch (OutOfStock $e) { } finally { } — Dart-এ on Type catch (e)।
Dart
class OutOfStock implements Exception {
  OutOfStock(this.product, this.have);
  final String product;
  final int have;
  @override
  String toString() => '$product: আছে মাত্র $have টা';
}

void sell(Map<String, int> stock, String item, int qty) {
  final have = stock[item] ?? 0;
  if (have < qty) throw OutOfStock(item, have);
  stock[item] = have - qty;
}

void main() {
  final stock = {'iPhone 12': 1};
  try {
    sell(stock, 'iPhone 12', 1);
    sell(stock, 'iPhone 12', 1);
  } on OutOfStock catch (e) {
    print('বিক্রি হলো না — $e');
  } finally {
    print('এখন স্টক: $stock');
  }
}

Project: দোকানের cart

নিজে করুন sealed class Discount বানান: Percent(5) আর Flat(500) — checkout যেকোনোটা নেবে।
Dart
enum Pay { cash, bkash }

class Item {
  const Item(this.name, this.price, [this.qty = 1]);
  final String name;
  final int price, qty;
  int get total => price * qty;
}

class Cart {
  final _items = <Item>[];
  void add(Item i) => _items.add(i);
  int get subtotal => _items.fold(0, (s, i) => s + i.total);
  int discount() => subtotal >= 20000 ? subtotal * 5 ~/ 100 : 0;

  String checkout(Pay pay) {
    final lines = [
      for (final i in _items) '${i.name.padRight(10)} ${i.qty} × ${i.price} = ${i.total}',
      'Subtotal: $subtotal',
      if (discount() > 0) 'ছাড় (৫%): -${discount()}',
      'দেবেন: ${subtotal - discount()} (${pay.name})',
    ];
    return lines.join('\n');
  }
}

void main() {
  final cart = Cart()
    ..add(const Item('Redmi 13', 17500))
    ..add(const Item('Charger', 650, 2))
    ..add(const Item('Cover', 250, 5));
  print(cart.checkout(Pay.bkash));
}
