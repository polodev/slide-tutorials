goroutine আর WaitGroup

কোনো function-এর আগে go লিখলেই সেটা আলাদা goroutine-এ একসাথে চলে। সব শেষ হওয়া পর্যন্ত অপেক্ষা করতে sync.WaitGroup।

PHP/Laravel-এ যেমন PHP-তে তিনটা API ডাকলে একটার পর একটা — মোট সময় যোগ হয়। Go-তে একসাথে চলে, সময় লাগে সবচেয়ে ধীরটার সমান।
💡 wg.Go(func) (Go 1.25+) — Add/Done নিজেই করে। পুরনো কোডে দেখবেন wg.Add(1) আর defer wg.Done()।
Go
package main

import (
	"fmt"
	"sync"
	"time"
)

func fetchBranch(name string, ms int) {
	time.Sleep(time.Duration(ms) * time.Millisecond) // ধরি API call
	fmt.Println("এসেছে:", name)
}

func main() {
	start := time.Now()
	var wg sync.WaitGroup
	for _, b := range []struct {
		name string
		ms   int
	}{{"Gulshan", 300}, {"Mirpur", 100}, {"Uttara", 200}} {
		wg.Go(func() { fetchBranch(b.name, b.ms) })
	}
	wg.Wait()
	fmt.Printf("সময় ~%dms (৬০০ না!)\n", time.Since(start).Round(100*time.Millisecond).Milliseconds())
}

Channel: goroutine-এর মধ্যে বার্তা

ch <- v পাঠায়, v := <-ch নেয়। পাঠানো আর নেওয়া একে অপরের জন্য অপেক্ষা করে — তাই lock ছাড়াই নিরাপদ।

💡 Go-র মূলমন্ত্র: "memory share করে যোগাযোগ কোরো না; যোগাযোগ করে memory share কোরো।" — channel দিয়ে data পাঠান।
Go
package main

import "fmt"

func branchTotal(name string, sales []int, out chan<- string) {
	sum := 0
	for _, s := range sales {
		sum += s
	}
	out <- fmt.Sprintf("%s: %d", name, sum)
}

func main() {
	out := make(chan string)
	go branchTotal("Mirpur", []int{18150, 21300, 18000}, out)
	go branchTotal("Uttara", []int{24150, 3450}, out)

	fmt.Println(<-out)
	fmt.Println(<-out)
}

Uttara: 27600
Mirpur: 57450

select: যেটা আগে আসে

select কয়েকটা channel-এর জন্য একসাথে অপেক্ষা করে — যেটা আগে তৈরি সেটা নেয়। time.After দিয়ে timeout।

Go
package main

import (
	"fmt"
	"time"
)

func api(name string, ms int) <-chan string {
	ch := make(chan string, 1)
	go func() {
		time.Sleep(time.Duration(ms) * time.Millisecond)
		ch <- name
	}()
	return ch
}

func main() {
	select {
	case r := <-api("bKash", 300):
		fmt.Println("আগে এলো:", r)
	case r := <-api("Nagad", 100):
		fmt.Println("আগে এলো:", r)
	}

	select {
	case r := <-api("slow-sms", 500):
		fmt.Println(r)
	case <-time.After(200 * time.Millisecond):
		fmt.Println("timeout — SMS পরে পাঠাব")
	}
}
আউটপুট
আগে এলো: Nagad
timeout — SMS পরে পাঠাব

Worker pool

১০০০টা invoice-এর SMS — একসাথে ১০০০ goroutine না, নির্দিষ্ট কয়েকজন worker। Job channel-এ কাজ, result channel-এ ফল।

PHP/Laravel-এ যেমন Laravel-এ php artisan queue:work কয়েকটা চালালে যা হয় — Go-তে এক program-এর ভিতরেই।
Go
package main

import (
	"fmt"
	"sync"
)

func main() {
	jobs := make(chan int)
	results := make(chan string)
	var wg sync.WaitGroup

	for w := 1; w <= 3; w++ {
		wg.Go(func() {
			for inv := range jobs {
				results <- fmt.Sprintf("worker %d → INV-%d SMS পাঠাল", w, inv)
			}
		})
	}
	go func() {
		for inv := 1001; inv <= 1006; inv++ {
			jobs <- inv
		}
		close(jobs)
	}()
	go func() { wg.Wait(); close(results) }()

	count := 0
	for range results {
		count++
	}
	fmt.Println("মোট SMS:", count, "— ৩ জন worker দিয়ে")
}
আউটপুট
মোট SMS: 6 — ৩ জন worker দিয়ে

context: timeout আর cancel

context কাজের সাথে সময়সীমা আর "বাতিল" সংকেত বহন করে। User request ছেড়ে দিলে database query-ও থামানো যায় — Go-র প্রতিটি web আর DB library context নেয়।

Go
package main

import (
	"context"
	"errors"
	"fmt"
	"time"
)

func slowReport(ctx context.Context) (string, error) {
	select {
	case <-time.After(500 * time.Millisecond):
		return "রিপোর্ট তৈরি", nil
	case <-ctx.Done():
		return "", ctx.Err()
	}
}

func main() {
	ctx, cancel := context.WithTimeout(context.Background(), 200*time.Millisecond)
	defer cancel()

	_, err := slowReport(ctx)
	fmt.Println("ফল:", err, "| deadline?", errors.Is(err, context.DeadlineExceeded))

	r, err := slowReport(context.Background())
	fmt.Println("ফল:", r, err)
}
আউটপুট
ফল: context deadline exceeded | deadline? true
ফল: রিপোর্ট তৈরি <nil>

Mutex: ভাগ করা data

অনেক goroutine একটা map বা সংখ্যা বদলালে sync.Mutex দিয়ে তালা দিতে হয়। go run -race ভুলে যাওয়া তালা ধরে ফেলে।

⚠️ Mutex ছাড়া একসাথে map-এ লিখলে Go program নিজেই থামিয়ে দেয়: fatal error: concurrent map writes।
Go
package main

import (
	"fmt"
	"sync"
)

type Stock struct {
	mu  sync.Mutex
	qty map[string]int
}

func (s *Stock) Sell(name string) {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.qty[name]--
}

func main() {
	s := &Stock{qty: map[string]int{"Charger": 100}}
	var wg sync.WaitGroup
	for range 40 {
		wg.Go(func() { s.Sell("Charger") })
	}
	wg.Wait()
	fmt.Println("বাকি Charger:", s.qty["Charger"])
}
আউটপুট
বাকি Charger: 60

Table-driven test

Go-তে test framework আলাদা install করতে হয় না: _test.go file, func TestXxx(t *testing.T), আর go test। প্রচলিত ধরন হলো table-driven — অনেক case একটা table-এ।

PHP/Laravel-এ যেমন Pest-এর dataset (->with([...]))-এর মতো। go test ./... পুরো project-এর সব test চালায়।
main.go
package main

func Vat(amount int) int { return amount * 15 / 100 }

func Discount(amount int) int {
	if amount >= 10000 {
		return amount / 10
	}
	return 0
}

func DeliveryCharge(district string) int {
	if district == "Dhaka" {
		return 60
	}
	return 120
}

func main() {}
main_test.go
package main

import "testing"

func TestDiscount(t *testing.T) {
	cases := []struct {
		name   string
		amount int
		want   int
	}{
		{"১০ হাজারের কম", 9999, 0},
		{"ঠিক ১০ হাজার", 10000, 1000},
		{"২০ হাজার", 20000, 2000},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			if got := Discount(tc.amount); got != tc.want {
				t.Errorf("Discount(%d) = %d, চাই %d", tc.amount, got, tc.want)
			}
		})
	}
}

func TestVat(t *testing.T) {
	if Vat(1000) != 150 {
		t.Fatal("VAT ১৫% হওয়ার কথা")
	}
}

Project: সব শাখার রিপোর্ট একসাথে

নিজে করুন Report গুলো channel দিয়ে সংগ্রহ করুন (Mutex ছাড়া), আর একটা test লিখুন যেখানে timeout ১ সেকেন্ড হলে তিন শাখাই আসে।
Go
package main

import (
	"context"
	"fmt"
	"slices"
	"sync"
	"time"
)

type Report struct {
	Branch string
	Total  int
	Err    error
}

func fetch(ctx context.Context, branch string, ms int, sales []int) Report {
	select {
	case <-time.After(time.Duration(ms) * time.Millisecond):
		sum := 0
		for _, s := range sales {
			sum += s
		}
		return Report{Branch: branch, Total: sum}
	case <-ctx.Done():
		return Report{Branch: branch, Err: ctx.Err()}
	}
}

func main() {
	ctx, cancel := context.WithTimeout(context.Background(), 250*time.Millisecond)
	defer cancel()

	branches := map[string]struct {
		ms    int
		sales []int
	}{
		"Mirpur":  {100, []int{18150, 2100, 21300}},
		"Uttara":  {150, []int{24150, 3450}},
		"Gulshan": {400, []int{48000, 21650}}, // খুব ধীর
	}

	var mu sync.Mutex
	var reports []Report
	var wg sync.WaitGroup
	for name, b := range branches {
		wg.Go(func() {
			r := fetch(ctx, name, b.ms, b.sales)
			mu.Lock()
			reports = append(reports, r)
			mu.Unlock()
		})
	}
	wg.Wait()

	slices.SortFunc(reports, func(a, b Report) int { return b.Total - a.Total })
	for _, r := range reports {
		if r.Err != nil {
			fmt.Printf("%-8s সময় শেষ (%v)\n", r.Branch, r.Err)
			continue
		}
		fmt.Printf("%-8s ৳%d\n", r.Branch, r.Total)
	}
}
