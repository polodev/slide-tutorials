struct আর method

PHP/Laravel-এ যেমন class Product { public string $name; ... public function stockValue() } — Go-তে data (struct) আর method আলাদা লেখা; (p Product) হলো PHP-র $this।
💡 বড় হাতের অক্ষরে শুরু নাম (Name, StockValue) package-এর বাইরে দেখা যায় = public। ছোট হাতের = private।
Go
package main

import "fmt"

type Product struct {
	Name  string
	Price int
	Stock int
}

func (p Product) StockValue() int {
	return p.Price * p.Stock
}

func main() {
	p := Product{Name: "Redmi 13", Price: 17500, Stock: 5}
	fmt.Println(p.Name, p.StockValue())
	fmt.Printf("%+v\n", p)
}
আউটপুট
Redmi 13 87500
{Name:Redmi 13 Price:17500 Stock:5}

Pointer receiver: নিজেকে বদলানো

Value receiver (p Product) কপি পায় — বদলালে আসলটা বদলায় না। বদলাতে চাইলে pointer receiver (p *Product)।

PHP/Laravel-এ যেমন PHP object সবসময় handle হিসেবে যায় (তাই বদলায়)। Go-তে struct ডিফল্টভাবে কপি হয় — বদলাতে * (pointer) লাগে।
Go
package main

import "fmt"

type Product struct {
	Name  string
	Stock int
}

func (p Product) SellCopy(qty int) { p.Stock -= qty } // কপি বদলাল
func (p *Product) Sell(qty int)    { p.Stock -= qty } // আসলটা বদলাল

func main() {
	p := Product{Name: "Vivo Y28", Stock: 4}
	p.SellCopy(1)
	fmt.Println("SellCopy-এর পর:", p.Stock)
	p.Sell(1)
	fmt.Println("Sell-এর পর:", p.Stock)
}
আউটপুট
SellCopy-এর পর: 4
Sell-এর পর: 3

Constructor function

Go-তে constructor নেই — প্রথা হলো NewXxx function, যা validation করে pointer ফেরত দেয়।

Go
package main

import (
	"errors"
	"fmt"
)

type Customer struct {
	Name  string
	Phone string
	Due   int
}

func NewCustomer(name, phone string) (*Customer, error) {
	if len(phone) != 11 {
		return nil, errors.New("ফোন নম্বর ১১ সংখ্যার হতে হবে")
	}
	return &Customer{Name: name, Phone: phone}, nil
}

func main() {
	c, err := NewCustomer("Karim", "01711000001")
	fmt.Println(c, err)
	_, err = NewCustomer("Nusrat", "0181")
	fmt.Println(err)
}
আউটপুট
&{Karim 01711000001 0} <nil>
ফোন নম্বর ১১ সংখ্যার হতে হবে

Embedding: inheritance-এর বদলে

একটা struct-এর ভিতরে আরেকটা struct নাম ছাড়া বসালে তার field আর method সরাসরি পাওয়া যায় — "is-a" না, "has-a"।

PHP/Laravel-এ যেমন Laravel model-এর trait (use HasTimestamps;)-এর মতো কাজ করে।
💡 Go-তে তারিখের format লেখা হয় একটা নির্দিষ্ট তারিখ দিয়ে: 2006-01-02 15:04:05 (১-২-৩-৪-৫-৬)।
Go
package main

import (
	"fmt"
	"time"
)

type Timestamps struct {
	CreatedAt time.Time
}

func (t *Timestamps) Touch() { t.CreatedAt = time.Date(2026, 10, 21, 10, 30, 0, 0, time.UTC) }

type Customer struct {
	Timestamps // embedded
	Name       string
}

func main() {
	c := Customer{Name: "Karim"}
	c.Touch() // Timestamps-এর method সরাসরি
	fmt.Println(c.Name, c.CreatedAt.Format("2006-01-02 15:04"))
}
আউটপুট
Karim 2026-10-21 10:30

Interface: নিজে থেকেই মেলে

Go-তে implements লেখা লাগে না। কোনো type-এর interface-এর সব method থাকলেই সে ঐ interface — একে বলে implicit interface।

PHP/Laravel-এ যেমন PHP-তে class Invoice implements Payable লিখতে হয়। Go-তে লেখার দরকারই নেই — test-এ fake বানানো খুব সহজ।
Go
package main

import "fmt"

type Payable interface {
	Amount() int
	Label() string
}

type Invoice struct{ No, Total int }
type ServiceBill struct{ Parts, Labour int }

func (i Invoice) Amount() int       { return i.Total }
func (i Invoice) Label() string     { return fmt.Sprintf("Invoice #%d", i.No) }
func (s ServiceBill) Amount() int   { return s.Parts + s.Labour }
func (s ServiceBill) Label() string { return "Service" }

func main() {
	items := []Payable{Invoice{101, 23500}, ServiceBill{1800, 500}}
	total := 0
	for _, it := range items {
		fmt.Printf("%-13s %6d\n", it.Label(), it.Amount())
		total += it.Amount()
	}
	fmt.Println("মোট:", total)
}
আউটপুট
Invoice #101   23500
Service         2300
মোট: 25800

Stringer আর type switch

String() string method থাকলে fmt সেটা দিয়েই প্রিন্ট করে (PHP-র __toString)। switch v := x.(type) দিয়ে আসল type দেখা।

Go
package main

import "fmt"

type Taka int

func (t Taka) String() string { return fmt.Sprintf("৳%d", int(t)) }

func describe(x any) string {
	switch v := x.(type) {
	case Taka:
		return "টাকা: " + v.String()
	case int:
		return fmt.Sprintf("সংখ্যা %d", v)
	case string:
		return "লেখা: " + v
	case nil:
		return "কিছু নেই"
	default:
		return fmt.Sprintf("অজানা %T", v)
	}
}

func main() {
	fmt.Println(Taka(17500))
	for _, x := range []any{Taka(650), 42, "Redmi", nil, 3.5} {
		fmt.Println(describe(x))
	}
}
আউটপুট
৳17500
টাকা: ৳650
সংখ্যা 42
লেখা: Redmi
কিছু নেই
অজানা float64

Generic

PHP/Laravel-এ যেমন Laravel Collection-এর filter()/sum() যেকোনো data-তে চলে — Go generic দিয়ে সেটাই, কিন্তু type-নিরাপদ।
Go
package main

import "fmt"

type Number interface{ ~int | ~float64 }

func Sum[T Number](items []T) T {
	var total T
	for _, x := range items {
		total += x
	}
	return total
}

func Filter[T any](items []T, keep func(T) bool) []T {
	var out []T
	for _, x := range items {
		if keep(x) {
			out = append(out, x)
		}
	}
	return out
}

func main() {
	prices := []int{17500, 650, 21000, 250}
	fmt.Println(Sum(prices), Sum([]float64{4.5, 3.9}))
	fmt.Println(Filter(prices, func(p int) bool { return p > 1000 }))
	fmt.Println(Filter([]string{"Dhaka", "Khulna", "Dinajpur"}, func(s string) bool { return s[0] == 'D' }))
}
আউটপুট
39400 8.4
[17500 21000]
[Dhaka Dinajpur]

Error wrapping: %w, errors.Is, errors.As

fmt.Errorf("...: %w", err) দিয়ে error-এর উপর প্রসঙ্গ যোগ হয়, ভিতরের error হারায় না। পরে errors.Is (নির্দিষ্ট error?) আর errors.As (নির্দিষ্ট type?) দিয়ে খোঁজা যায়।

PHP/Laravel-এ যেমন PHP-র exception chaining (new Exception('...', 0, $previous)) আর instanceof-এর মতো।
Go
package main

import (
	"errors"
	"fmt"
)

var ErrNotFound = errors.New("পাওয়া যায়নি")

type StockError struct {
	Product string
	Have    int
}

func (e *StockError) Error() string {
	return fmt.Sprintf("%s: আছে মাত্র %d", e.Product, e.Have)
}

func findProduct(id int) error {
	if id > 100 {
		return ErrNotFound
	}
	return &StockError{Product: "iPhone 12", Have: 0}
}

func sell(id int) error {
	if err := findProduct(id); err != nil {
		return fmt.Errorf("sell %d: %w", id, err)
	}
	return nil
}

func main() {
	err := sell(999)
	fmt.Println(err, "| not found?", errors.Is(err, ErrNotFound))

	err = sell(4)
	var se *StockError
	if errors.As(err, &se) {
		fmt.Println(err, "| product:", se.Product)
	}
}
আউটপুট
sell 999: পাওয়া যায়নি | not found? true
sell 4: iPhone 12: আছে মাত্র 0 | product: iPhone 12

JSON আর struct tag

Struct-এর field-এর পাশে `json:"name"` tag দিয়ে JSON-এর key ঠিক করি। omitempty — ফাঁকা হলে বাদ, - — কখনো না।

PHP/Laravel-এ যেমন json_encode() আর Laravel model-এর $hidden = ['cost_price'] — Go-তে `json:"-"`।
Go
package main

import (
	"encoding/json"
	"fmt"
)

type Product struct {
	ID        int      `json:"id"`
	Name      string   `json:"name"`
	Price     int      `json:"price"`
	Colors    []string `json:"colors,omitempty"`
	CostPrice int      `json:"-"`
}

func main() {
	p := Product{ID: 7, Name: "Redmi 13", Price: 17500, CostPrice: 15800}
	out, _ := json.Marshal(p)
	fmt.Println(string(out))

	var in Product
	err := json.Unmarshal([]byte(`{"id":2,"name":"Galaxy A15","price":21000,"colors":["Blue"]}`), &in)
	fmt.Printf("%+v %v\n", in, err)

	err = json.Unmarshal([]byte(`{"id":"two"}`), &in)
	fmt.Println(err)
}
আউটপুট
{"id":7,"name":"Redmi 13","price":17500}
{ID:2 Name:Galaxy A15 Price:21000 Colors:[Blue] CostPrice:0} <nil>
json: cannot unmarshal string into Go struct field Product.id of type int

Project: Inventory

নিজে করুন Inventory-কে JSON-এ save আর load করুন (encoding/json + os.WriteFile)।
Go
package main

import (
	"errors"
	"fmt"
	"slices"
	"strings"
)

type Product struct {
	Name  string
	Price int
	Stock int
}

type Inventory struct{ items map[string]*Product }

var ErrNoStock = errors.New("স্টক নেই")

func NewInventory() *Inventory { return &Inventory{items: map[string]*Product{}} }

func (inv *Inventory) Add(name string, price, qty int) {
	if p, ok := inv.items[name]; ok {
		p.Stock += qty
		return
	}
	inv.items[name] = &Product{name, price, qty}
}

func (inv *Inventory) Sell(name string, qty int) (int, error) {
	p, ok := inv.items[name]
	if !ok {
		return 0, fmt.Errorf("%q: পাওয়া যায়নি", name)
	}
	if p.Stock < qty {
		return 0, fmt.Errorf("%s (আছে %d): %w", name, p.Stock, ErrNoStock)
	}
	p.Stock -= qty
	return p.Price * qty, nil
}

func (inv *Inventory) Report() string {
	names := make([]string, 0, len(inv.items))
	for n := range inv.items {
		names = append(names, n)
	}
	slices.Sort(names)
	var b strings.Builder
	for _, n := range names {
		p := inv.items[n]
		fmt.Fprintf(&b, "%-12s %3dটা  ৳%d\n", n, p.Stock, p.Price*p.Stock)
	}
	return b.String()
}

func main() {
	inv := NewInventory()
	inv.Add("Redmi 13", 17500, 5)
	inv.Add("Galaxy A15", 21000, 2)
	inv.Add("Redmi 13", 17500, 1)

	for _, s := range []struct {
		name string
		qty  int
	}{{"Redmi 13", 2}, {"Galaxy A15", 3}, {"Nokia", 1}} {
		if amount, err := inv.Sell(s.name, s.qty); err != nil {
			fmt.Println("✗", err, "| no stock?", errors.Is(err, ErrNoStock))
		} else {
			fmt.Println("✓", s.name, "বিক্রি:", amount)
		}
	}
	fmt.Print(inv.Report())
}
আউটপুট
✓ Redmi 13 বিক্রি: 35000
✗ Galaxy A15 (আছে 2): স্টক নেই | no stock? true
✗ "Nokia": পাওয়া যায়নি | no stock? false
Galaxy A15     2টা  ৳42000
Redmi 13       4টা  ৳70000
