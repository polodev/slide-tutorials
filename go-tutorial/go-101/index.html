fmt.Printf: সাজিয়ে প্রিন্ট

PHP/Laravel-এ যেমন PHP-র printf()/sprintf() একই — %v হলো Go-র "যেকোনো মান", %T type দেখায়।
Go
package main

import "fmt"

func main() {
	name := "Rahim"
	price := 1250.5
	qty := 3

	fmt.Printf("Customer: %s\n", name)
	fmt.Printf("Total: %.2f টাকা\n", price*float64(qty))
	fmt.Printf("[%8s] [%-8s]\n", "ডানে", "বামে")
	fmt.Printf("%v | %T | %q\n", qty, price, name)
	line := fmt.Sprintf("%s × %d", name, qty)
	fmt.Println(line)
}
আউটপুট
Customer: Rahim
Total: 3751.50 টাকা
[    ডানে] [বামে    ]
3 | float64 | "Rahim"
Rahim × 3

var, := আর zero value

:= দিয়ে variable বানানো আর মান দেওয়া একসাথে (শুধু function-এর ভিতরে)। মান না দিলে Go নিজেই "zero value" বসায় — null নেই।

PHP/Laravel-এ যেমন PHP-তে মান না দেওয়া variable null/undefined। Go-তে int হলে 0, string হলে "" — কখনো "নেই" হয় না।
Go
package main

import "fmt"

func main() {
	var shop string = "Rahim Mobile"
	balance := 500 // type অনুমান: int
	balance += 200

	var stock int   // মান দিইনি → 0
	var note string // → ""
	var paid bool   // → false

	fmt.Println(shop, balance)
	fmt.Printf("stock=%d note=%q paid=%t\n", stock, note, paid)
}
আউটপুট
Rahim Mobile 700
stock=0 note="" paid=false

ব্যবহার না করা variable = error

Go কঠোর: variable বানিয়ে ব্যবহার না করলে, বা package import করে ব্যবহার না করলে — compile হবে না। কোড পরিষ্কার থাকে।

💡 সাময়িকভাবে কিছু উপেক্ষা করতে _ ব্যবহার করুন: _ = total।
Go
package main

import (
	"fmt"
	"os"
)

func main() {
	total := 1500
	fmt.Println("হিসাব")
}

Type আর conversion

আলাদা type মেশানো যায় না — নিজে বদলাতে হয়: float64(qty)। Integer ভাগে দশমিক বাদ।

⚠️ string(65) দিলে "65" না, "A" হয় (অক্ষর কোড)! সংখ্যাকে লেখা বানাতে সবসময় strconv.Itoa বা fmt.Sprint।
Go
package main

import (
	"fmt"
	"strconv"
)

func main() {
	a, b := 7, 2
	fmt.Println(a/b, a%b, float64(a)/float64(b))

	price := 17500
	vat := float64(price) * 0.15
	fmt.Printf("VAT: %.0f\n", vat)

	n, _ := strconv.Atoi("250")
	s := strconv.Itoa(price)
	fmt.Println(n+1, s+" টাকা")
}
আউটপুট
3 1 3.5
VAT: 2625
251 17500 টাকা

const আর iota

PHP/Laravel-এ যেমন iota নিজে থেকে ০, ১, ২ ... বসায় — PHP enum-এর মতো ক্রমিক মানের জন্য।
Go
package main

import "fmt"

const VatPercent = 15
const ShopName = "Rahim Mobile"

type Status int

const (
	Pending  Status = iota // 0
	Paid                   // 1
	Returned               // 2
)

func main() {
	fmt.Println(ShopName, VatPercent)
	fmt.Println(Pending, Paid, Returned)
}
আউটপুট
Rahim Mobile 15
0 1 2

String, byte আর rune

Go string হলো UTF-8 byte। len() byte গোনে। অক্ষর (rune) ধরে চলতে for range বা utf8.RuneCountInString।

PHP/Laravel-এ যেমন strlen/mb_strlen → len/utf8.RuneCountInString; explode → strings.Split।
Go
package main

import (
	"fmt"
	"strings"
	"unicode/utf8"
)

func main() {
	city := "ঢাকা"
	fmt.Println(len(city), utf8.RuneCountInString(city))
	for i, r := range city {
		fmt.Printf("%d:%c ", i, r)
	}
	fmt.Println()

	model := "  Redmi Note 13  "
	fmt.Println(strings.TrimSpace(model), strings.ToUpper("redmi"), strings.Contains(model, "Note"))
	fmt.Println(strings.Split("Mirpur,Uttara,Gulshan", ","))
}
আউটপুট
12 4
0:ঢ 3:া 6:ক 9:া 
Redmi Note 13 REDMI true
[Mirpur Uttara Gulshan]

if — ছোট statement সহ

শর্তের চারপাশে bracket নেই, কিন্তু {} বাধ্যতামূলক। if-এর আগে ছোট statement লেখা যায়। Ternary (?:) নেই।

Go
package main

import "fmt"

func main() {
	amount := 1800
	if amount >= 1500 {
		fmt.Println("ফ্রি ডেলিভারি")
	} else if amount >= 1000 {
		fmt.Println("ডেলিভারি ৩০")
	} else {
		fmt.Println("ডেলিভারি ৬০")
	}

	if discount := amount / 10; discount > 100 {
		fmt.Println("ছাড়:", discount)
	}
}
আউটপুট
ফ্রি ডেলিভারি
ছাড়: 180

for — Go-র একমাত্র loop

while নেই, foreach নেই — সব for। range দিয়ে slice/map/string ঘোরা।

PHP/Laravel-এ যেমন foreach ($cities as $i => $city) → for i, city := range cities।
Go
package main

import "fmt"

func main() {
	for i := 1; i <= 3; i++ {
		fmt.Printf("7 × %d = %d\n", i, 7*i)
	}

	seconds := 3
	for seconds > 0 { // while-এর মতো
		fmt.Print(seconds, "... ")
		seconds--
	}
	fmt.Println()

	for i, city := range []string{"Dhaka", "Khulna", "Sylhet"} {
		fmt.Println(i, city)
	}
}
আউটপুট

7 × 3 = 21
3... 2... 1... 
0 Dhaka
1 Khulna
2 Sylhet

switch

break লিখতে হয় না — মিল হলে শুধু সেই case চলে। শর্ত ছাড়া switch লিখলে if-else-এর মতো।

Go
package main

import "fmt"

func grade(marks int) string {
	switch {
	case marks >= 80:
		return "A+"
	case marks >= 70:
		return "A"
	case marks >= 33:
		return "পাস"
	default:
		return "F"
	}
}

func main() {
	for _, m := range []int{95, 72, 48, 20} {
		fmt.Println(m, grade(m))
	}
	switch day := "Friday"; day {
	case "Friday":
		fmt.Println("ছুটি")
	case "Saturday", "Sunday":
		fmt.Println("কাজের দিন")
	}
}
আউটপুট
95 A+
72 A
48 পাস
20 F
ছুটি

Function আর একাধিক return

PHP/Laravel-এ যেমন PHP-তে কয়েকটা মান ফেরত দিতে array লাগে; Go সরাসরি (int, int) ফেরত দেয়।
Go
package main

import "fmt"

func total(price, qty int) int {
	return price * qty
}

func minMax(nums []int) (int, int) {
	lo, hi := nums[0], nums[0]
	for _, n := range nums {
		lo, hi = min(lo, n), max(hi, n)
	}
	return lo, hi
}

func main() {
	fmt.Println(total(450, 4))
	lo, hi := minMax([]int{18150, 2100, 48000, 1300})
	fmt.Println("ছোট", lo, "বড়", hi)
}
আউটপুট
1800
ছোট 1300 বড় 48000

error: Go-র নিয়ম

ব্যর্থ হতে পারে এমন function শেষে একটা error ফেরত দেয়। ডাকার পরেই if err != nil — Go কোডে এটা সবচেয়ে বেশি দেখবেন।

PHP/Laravel-এ যেমন PHP-র throw/catch নেই। Rust-এর Result-এর মতো — error উপেক্ষা করা কঠিন, কারণ চোখের সামনে থাকে।
Go
package main

import (
	"errors"
	"fmt"
	"strconv"
)

func withdraw(balance, amount int) (int, error) {
	if amount > balance {
		return balance, fmt.Errorf("ব্যালেন্স কম: আছে %d, চাইছেন %d", balance, amount)
	}
	return balance - amount, nil
}

var ErrZero = errors.New("০ টাকা তোলা যায় না")

func main() {
	if _, err := strconv.Atoi("দুইশো"); err != nil {
		fmt.Println("ভুল:", err)
	}
	left, err := withdraw(1000, 5000)
	if err != nil {
		fmt.Println("ব্যর্থ:", err)
	}
	fmt.Println("বাকি:", left, ErrZero)
}
আউটপুট
ভুল: strconv.Atoi: parsing "দুইশো": invalid syntax
ব্যর্থ: ব্যালেন্স কম: আছে 1000, চাইছেন 5000
বাকি: 1000 ০ টাকা তোলা যায় না

package main

Slice: বাড়ানো যায় এমন list

PHP/Laravel-এ যেমন $brands[] = 'Walton' → brands = append(brands, "Walton") — append নতুন slice ফেরত দেয়, তাই আবার বসাতে হয়।
Go
package main

import (
	"fmt"
	"slices"
	"strings"
)

func main() {
	brands := []string{"Samsung", "Xiaomi"}
	brands = append(brands, "Walton", "Vivo")

	fmt.Println(brands, len(brands))
	fmt.Println(brands[1:3]) // index ১ থেকে ২
	fmt.Println(slices.Contains(brands, "Walton"))
	slices.Sort(brands)
	fmt.Println(strings.Join(brands, " | "))
}
আউটপুট
[Samsung Xiaomi Walton Vivo] 4
[Xiaomi Walton]
true
Samsung | Vivo | Walton | Xiaomi

Map আর "comma ok"

⚠️ Map-এর range-এর ক্রম প্রতিবার আলাদা হতে পারে। ক্রম লাগলে key নিয়ে slices.Sort করুন।
PHP/Laravel-এ যেমন $stock['Nokia'] ?? 0 আর isset() — Go-তে একসাথে: qty, ok := stock["Nokia"]।
Go
package main

import "fmt"

func main() {
	stock := map[string]int{
		"Redmi 13":   5,
		"Galaxy A15": 2,
	}
	stock["Vivo Y28"] = 4
	stock["Redmi 13"]--

	qty, ok := stock["Nokia"]
	fmt.Println(qty, ok) // নেই: 0 false

	delete(stock, "Galaxy A15")
	for name, n := range stock {
		if n > 3 {
			fmt.Println(name, n)
		}
	}
	fmt.Println(len(stock))
}
আউটপুট
0 false
Redmi 13 4
Vivo Y28 4
2

defer: শেষে চালাও

defer দেওয়া কাজ function শেষ হওয়ার ঠিক আগে চলে — file বন্ধ, lock খোলা, connection ফেরত। একাধিক defer উল্টো ক্রমে চলে।

PHP/Laravel-এ যেমন PHP-র try { } finally { }-এর কাজ, কিন্তু যেখানে resource খোলেন ঠিক তার পাশেই লেখা যায়।
Go
package main

import "fmt"

func sell() {
	fmt.Println("lock নিলাম")
	defer fmt.Println("lock ছাড়লাম") // যাই হোক, শেষে চলবে
	defer fmt.Println("log লিখলাম")
	fmt.Println("স্টক কমালাম")
}

func main() {
	sell()
}
আউটপুট
lock নিলাম
স্টক কমালাম
log লিখলাম
lock ছাড়লাম

Project: টাকার নোট ভাঙানো

নিজে করুন একটা map[int]int-এ রাখুন কোন নোট কয়টা, আর ভুল input-এ আবার জিজ্ঞেস করুন (for loop)।
Go
package main

import (
	"bufio"
	"fmt"
	"os"
	"strconv"
	"strings"
)

func main() {
	notes := []int{1000, 500, 200, 100, 50, 20, 10, 5, 2, 1}
	fmt.Print("কত টাকা ফেরত? ")
	line, _ := bufio.NewReader(os.Stdin).ReadString('\n')
	amount, err := strconv.Atoi(strings.TrimSpace(line))
	if err != nil {
		fmt.Println("সংখ্যা লিখুন")
		return
	}
	fmt.Println()
	total := 0
	for _, note := range notes {
		if count := amount / note; count > 0 {
			fmt.Printf("%5d টাকা × %d\n", note, count)
			amount %= note
			total += count
		}
	}
	fmt.Println("মোট নোট:", total)
}

   20 টাকা × 1
