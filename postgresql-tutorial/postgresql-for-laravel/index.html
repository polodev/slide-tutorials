    customer_id bigint REFERENCES customers (id),     -- walk-in হলে NULL

    qty        integer NOT NULL,

);

পাঠ ২

Migration কোন SQL বানায়

  • id() → bigserial primary key
  • enum() → varchar + CHECK (আসল enum type না)
  • jsonb() → আসল jsonb; json() দিলে শুধু json
  • timestampsTz() → timestamp(0) with time zone
Laravel migration → PostgreSQL
use Illuminate\Database\Schema\Blueprint;

$queries = DB::connection()->pretend(fn () => DB::schema()->create('repairs', function (Blueprint $t) {
    $t->id();
    $t->foreignId('customer_id')->constrained();
    $t->string('device');
    $t->enum('status', ['received', 'repairing', 'delivered'])->default('received');
    $t->boolean('warranty')->default(false);
    $t->jsonb('parts')->nullable();
    $t->decimal('charge', 10, 2);
    $t->timestampsTz();
}));
foreach ($queries as $q) echo $q['query'], ";\n";
Laravel চালিয়ে পাওয়া আউটপুট
create table "repairs" ("id" bigserial not null primary key, "customer_id" bigint not null, "device" varchar(255) not null, "status" varchar(255) check ("status" in ('received', 'repairing', 'delivered')) not null default 'received', "warranty" boolean not null default '0', "parts" jsonb null, "charge" decimal(10, 2) not null, "created_at" timestamp(0) with time zone null, "updated_at" timestamp(0) with time zone null);
alter table "repairs" add constraint "repairs_customer_id_foreign" foreign key ("customer_id") references "customers" ("id");
পাঠ ৩

LIKE এখন case-sensitive

⚠️ MySQL থেকে আসা app-এ search box হঠাৎ কিছু খুঁজে না পেলে প্রথমে এটা দেখুন। whereLike() দুই database-এই ঠিক আচরণ করে।
Laravel
echo "like:  ", Product::where('name', 'like', 'redmi%')->count(), "\n";
echo "ilike: ", Product::where('name', 'ilike', 'redmi%')->count(), "\n";

$q = Product::whereLike('name', 'redmi%');                       // Laravel 11+
echo "whereLike: ", $q->count(), "  →  ", $q->toRawSql(), "\n";
$q = Product::whereLike('name', 'redmi%', caseSensitive: true);
echo "caseSensitive: ", $q->count(), "  →  ", $q->toRawSql();
Laravel চালিয়ে পাওয়া আউটপুট
like:  0
ilike: 2
whereLike: 2  →  select * from "products" where "name"::text ilike 'redmi%'
caseSensitive: 0  →  select * from "products" where "name"::text like 'redmi%'

GROUP BY: MySQL-এ চলত, এখানে চলে না

try {
    Sale::selectRaw('branch_id, invoice_no, sum(total) as revenue')->groupBy('branch_id')->get();
} catch (Illuminate\Database\QueryException $e) {
    echo "SQLSTATE ", $e->errorInfo[0], ":\n", explode("\n", $e->errorInfo[2])[0], "\n\n";
}

$rows = Sale::selectRaw('branch_id, count(*) as invoices, sum(total) as revenue')
    ->groupBy('branch_id')->orderBy('branch_id')->get();
echo $rows->toJson();

পাঠ ৫

Sequence-এর ফাঁদ: seed-এর পর duplicate key

Laravel-এর id() হলো bigserial — একটা sequence থেকে পরের নম্বর নেয়। Seeder-এ id হাতে দিলে sequence জানে না, পরের create() একই id নিতে চায়।

💡 সমাধান: seeder-এ id দেবেন না; অথবা import-এর পর setval চালান। MySQL-এর AUTO_INCREMENT নিজেই এগিয়ে যায়, তাই এই সমস্যা সেখানে নেই।
Laravel
use Illuminate\Database\Schema\Blueprint;
DB::schema()->create('brands', fn (Blueprint $t) => [$t->id(), $t->string('name')]);

DB::table('brands')->insert([['id' => 1, 'name' => 'Xiaomi'], ['id' => 2, 'name' => 'Samsung']]);
try {
    DB::table('brands')->insert(['name' => 'Vivo']);
} catch (Illuminate\Database\QueryException $e) {
    echo explode("\n", $e->errorInfo[2])[0], "\n";
}

DB::statement("SELECT setval(pg_get_serial_sequence('brands', 'id'), (SELECT max(id) FROM brands))");
$id = DB::table('brands')->insertGetId(['name' => 'Vivo']);
echo "setval-এর পর নতুন id: {$id}";
Laravel চালিয়ে পাওয়া আউটপুট
ERROR:  duplicate key value violates unique constraint "brands_pkey"
setval-এর পর নতুন id: 3
পাঠ ৬

Aborted transaction

PostgreSQL-এ transaction-এর ভিতরে একটা query ব্যর্থ হলে, exception catch করে চালিয়ে গেলেও পরের সব query ব্যর্থ হয় — যতক্ষণ না rollback।

💡 Transaction-এর ভিতরে ভুল হওয়া স্বাভাবিক হলে (যেমন duplicate) আগেই যাচাই করুন, insertOrIgnore()/upsert() দিন, অথবা ভিতরে nested DB::transaction() (SAVEPOINT) ব্যবহার করুন।
Laravel
try {
    DB::transaction(function () {
        try {
            DB::insert("insert into customers (name, phone) values ('Dup', '01711000001')");   // duplicate phone
        } catch (Illuminate\Database\QueryException $e) {
            echo "১ম ব্যর্থ, catch করে চালিয়ে গেলাম...\n";
        }
        DB::table('customers')->count();                                                     // এটাও ব্যর্থ!
    });
} catch (Illuminate\Database\QueryException $e) {
    echo "২য়: SQLSTATE ", $e->errorInfo[0], " — ", explode("\n", $e->errorInfo[2])[0];
}
Laravel চালিয়ে পাওয়া আউটপুট
১ম ব্যর্থ, catch করে চালিয়ে গেলাম...
২য়: SQLSTATE 25P02 — ERROR:  current transaction is aborted, commands ignored until end of transaction block
পাঠ ৭

RETURNING: create আর insertGetId

PostgreSQL-এ নতুন id আসে একই query-র returning "id" থেকে — আলাদা LAST_INSERT_ID() লাগে না।

Laravel
queries(function () {
    Customer::create(['name' => 'Jamal', 'phone' => '01700000101']);
    DB::table('customers')->insertGetId(['name' => 'Kamal', 'phone' => '01700000102']);
});
Laravel চালিয়ে পাওয়া আউটপুট
[1] insert into "customers" ("name", "phone") values ('Jamal', '01700000101') returning "id"
[2] insert into "customers" ("name", "phone") values ('Kamal', '01700000102') returning "id"
→ মোট query: 2
পাঠ ৮

upsert → ON CONFLICT

⚠️ PostgreSQL-এ uniqueBy-এর column-এ সত্যিকারের unique index থাকতেই হবে, নইলে error। MySQL কোনো unique key পেলেই চলে।
PostgreSQL
CREATE UNIQUE INDEX products_name_unique ON products (name);
Laravel
queries(fn () => Product::upsert([
    ['category_id' => 1, 'name' => 'Redmi 13',  'brand' => 'Xiaomi', 'cost' => 15500, 'price' => 17200, 'stock' => 5],
    ['category_id' => 1, 'name' => 'Redmi 14C', 'brand' => 'Xiaomi', 'cost' => 13000, 'price' => 14500, 'stock' => 6],
], uniqueBy: ['name'], update: ['cost', 'price']));

queries(fn () => Customer::insertOrIgnore(['name' => 'Karim again', 'phone' => '01711000001']));
Laravel চালিয়ে পাওয়া আউটপুট
[1] insert into "products" ("brand", "category_id", "cost", "name", "price", "stock") values ('Xiaomi', 1, 15500, 'Redmi 13', 17200, 5), ('Xiaomi', 1, 13000, 'Redmi 14C', 14500, 6) on conflict ("name") do update set "cost" = "excluded"."cost", "price" = "excluded"."price"
→ মোট query: 1
[1] insert into "customers" ("name", "phone") values ('Karim again', '01711000001') on conflict do nothing
→ মোট query: 1
পাঠ ৯

JSONB Eloquent দিয়ে

⚠️ শেষ query-তে ->> লেখা (text) ফেরত দেয়, তাই তুলনা হয় অক্ষর ধরে: '10' >= '8' মিথ্যা! সংখ্যায় তুলনা করতে cast দিন: whereRaw("(specs->>'ram')::int >= ?", [8])।
💡 whereJsonContains → @>, যেটা GIN index ব্যবহার করে। ->> দিয়ে তুলনা index পায় না — বারবার লাগলে expression index দিন।
PostgreSQL
ALTER TABLE products ADD COLUMN specs jsonb NOT NULL DEFAULT '{}';
UPDATE products SET specs = '{"ram": 8, "colors": ["Black", "Green"], "5g": false}' WHERE name = 'Redmi Note 13';
UPDATE products SET specs = '{"ram": 4, "colors": ["Black"], "5g": true}'          WHERE name = 'iPhone 12';
CREATE INDEX products_specs_gin ON products USING gin (specs);
Laravel
$q = Product::whereJsonContains('specs->colors', 'Black');
echo $q->toRawSql(), "\n", $q->pluck('name')->join(', '), "\n\n";

$q = Product::where('specs->5g', true);
echo $q->toRawSql(), "\n", $q->pluck('name')->join(', '), "\n\n";

$q = Product::where('specs->ram', '>=', 8);
echo $q->toRawSql(), "\n", $q->pluck('name')->join(', ');
Laravel চালিয়ে পাওয়া আউটপুট
select * from "products" where ("specs"->'colors')::jsonb @> '"Black"'
Redmi Note 13, iPhone 12

select * from "products" where ("specs"->'5g')::jsonb = 'true'::jsonb
iPhone 12

select * from "products" where "specs"->>'ram' >= 8
Redmi Note 13
পাঠ ১০

Full-text: whereFullText

PHP/Laravel-এ যেমন Migration-এ $table->fullText(['name', 'brand']); দিলে pgsql-এ GIN index বানায় একই to_tsvector expression-এ — তাই query index পায়।
Laravel
$q = Product::whereFullText(['name', 'brand'], 'galaxy');
echo $q->toRawSql(), "\n", $q->pluck('name')->join(', ');
Laravel চালিয়ে পাওয়া আউটপুট
select * from "products" where (to_tsvector('english', "name") || to_tsvector('english', "brand")) @@ plainto_tsquery('english', 'galaxy')
Galaxy A15
পাঠ ১১

lockForUpdate আর queue

PHP/Laravel-এ যেমন Laravel-এর database queue PostgreSQL-এ for update skip locked দিয়ে job নেয় — একাধিক worker নিরাপদে।
Laravel
queries(fn () => DB::transaction(function () {
    $p = Product::lockForUpdate()->find(4);
    $p->decrement('stock');
}));

echo "\nlock + skip locked:\n";
echo DB::table('jobs_demo')->lock('for update skip locked')->limit(1)->toRawSql();
Laravel চালিয়ে পাওয়া আউটপুট
[1] select * from "products" where "products"."id" = 4 limit 1 for update
[2] update "products" set "stock" = "stock" - 1 where "id" = 4
→ মোট query: 2

lock + skip locked:
select * from "jobs_demo" limit 1 for update skip locked

MySQL থেকে PostgreSQL-এ সরানোর চেকলিস্ট

দেখুনকেন
like দিয়ে searchCase-sensitive → whereLike() বা ilike
groupBy + selectRawSELECT-এর সব column GROUP BY-তে
Raw SQL-এ backtick, IFNULL, DATE_FORMAT→ double quote, COALESCE, to_char
Seeder-এ হাতে idশেষে setval
Transaction-এ catch করে চালিয়ে যাওয়াAborted transaction → SAVEPOINT/আগাম যাচাই
upsertuniqueBy column-এ unique index
boolean, numericCast ঠিক আছে কিনা (string/bool)
json columnjsonb + GIN index
নিজে করুন নিজের Laravel project PostgreSQL-এ চালান (Docker দিয়ে), test suite চালিয়ে দেখুন কোন query ভাঙে — উপরের তালিকা মিলিয়ে ঠিক করুন।
