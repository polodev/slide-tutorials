MySQL টিউটোরিয়ালের একই দোকান। পার্থক্য: identity column, numeric, timestamp, আর foreign key column-এর পাশেই REFERENCES।

⚠️ PostgreSQL foreign key column-এ নিজে থেকে index বানায় না (MySQL বানায়)। JOIN দ্রুত করতে নিজে দিন: CREATE INDEX ON sales (customer_id);
PostgreSQL — schema
CREATE TABLE branches (
    id   bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name text NOT NULL
);

CREATE TABLE customers (
    id       bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name     text NOT NULL,
    phone    varchar(15) NOT NULL UNIQUE,
    district text
);

CREATE TABLE categories (
    id   bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    name text NOT NULL
);

CREATE TABLE products (
    id          bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    category_id bigint NOT NULL REFERENCES categories (id),
    name        text NOT NULL,
    brand       text NOT NULL,
    cost        numeric(10,2) NOT NULL,
    price       numeric(10,2) NOT NULL,
    stock       integer NOT NULL DEFAULT 0 CHECK (stock >= 0)
);

CREATE TABLE employees (
    id        bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    branch_id bigint NOT NULL REFERENCES branches (id),
    name      text NOT NULL
);

CREATE TABLE sales (
    id          bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    invoice_no  text NOT NULL UNIQUE,
    branch_id   bigint NOT NULL REFERENCES branches (id),
    employee_id bigint NOT NULL REFERENCES employees (id),
    customer_id bigint REFERENCES customers (id),     -- walk-in হলে NULL
    sold_at     timestamp NOT NULL,
    total       numeric(10,2) NOT NULL
);

CREATE TABLE sale_items (
    id         bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    sale_id    bigint NOT NULL REFERENCES sales (id) ON DELETE CASCADE,
    product_id bigint NOT NULL REFERENCES products (id),
    qty        integer NOT NULL,
    unit_price numeric(10,2) NOT NULL
);

কঠোর GROUP BY

💡 ব্যতিক্রম: primary key দিয়ে GROUP BY করলে ঐ table-এর বাকি column নিজেই চলে — GROUP BY b.id দিলে b.name লেখা যায়।

FILTER: এক query-তে অনেক শর্ত

count(*) FILTER (WHERE ...) — একই দলে আলাদা শর্তের হিসাব। MySQL-এ এর জন্য SUM(CASE WHEN ...) লিখতে হয়।

PostgreSQL
SELECT b.name AS branch,
       count(*)                                             AS invoices,
       count(*) FILTER (WHERE s.customer_id IS NULL)        AS walk_in,
       sum(s.total) FILTER (WHERE s.total >= 20000)         AS big_sales,
       round(avg(s.total) FILTER (WHERE s.total < 20000))   AS avg_small
FROM sales s
JOIN branches b ON b.id = s.branch_id
GROUP BY b.id
ORDER BY b.id;

 Gulshan |        2 |       0 |  69650.00 |

string_agg: invoice-এর সব product এক লাইনে

SELECT s.invoice_no,
       string_agg(p.name, ', ' ORDER BY p.name) AS products,
       sum(si.qty)                              AS items
FROM sales s

JOIN products p ON p.id = si.product_id

LIMIT 5;

array_agg: আসল array

PHP/Laravel-এ যেমন Laravel-এ pgsql array column পড়লে {INV-1001,INV-1005} আকারের string আসে — JSON চাইলে json_agg() দিন, PHP-তে json_decode।
PostgreSQL
SELECT e.name AS employee,
       array_agg(s.invoice_no ORDER BY s.sold_at) AS invoices,
       cardinality(array_agg(s.id))               AS how_many
FROM employees e
JOIN sales s ON s.employee_id = e.id
GROUP BY e.id
ORDER BY how_many DESC;

DISTINCT ON: প্রতি দলের প্রথম row

"প্রতিটি customer-এর সর্বশেষ কেনা" — MySQL-এ window function বা subquery লাগে। PostgreSQL-এ DISTINCT ON।

💡 নিয়ম: ORDER BY শুরু হবে DISTINCT ON-এর column দিয়ে, তারপর কোন row "প্রথম" সেটা ঠিক করার ক্রম।
PostgreSQL
SELECT DISTINCT ON (c.id)
       c.name, s.invoice_no, s.sold_at::date AS last_visit, s.total
FROM customers c
JOIN sales s ON s.customer_id = c.id
ORDER BY c.id, s.sold_at DESC;

LATERAL: প্রতি row-এর জন্য subquery

JOIN LATERAL-এর ভিতরের subquery বাইরের row দেখতে পায়। "প্রতিটি শাখার সবচেয়ে বড় ২টা invoice"।

PostgreSQL
SELECT b.name AS branch, top.invoice_no, top.total
FROM branches b
JOIN LATERAL (
    SELECT invoice_no, total
    FROM sales s
    WHERE s.branch_id = b.id
    ORDER BY total DESC
    LIMIT 2
) AS top ON true
ORDER BY b.name, top.total DESC;

PHP/Laravel-এ যেমন Laravel-এর upsert() pgsql-এ ঠিক এই ON CONFLICT ... DO UPDATE বানায়। insertOrIgnore() → ON CONFLICT DO NOTHING।

CREATE VIEW branch_daily_sales AS

Project: সপ্তাহের রিপোর্ট

নিজে করুন FILTER দিয়ে একই রিপোর্টে আরেকটা column যোগ করুন: শুধু Smartphone category-র লাভ।
PostgreSQL
SELECT b.name AS branch,
       e.name AS employee,
       count(DISTINCT s.id)                                  AS invoices,
       sum(si.qty * si.unit_price)                           AS revenue,
       sum(si.qty * (si.unit_price - p.cost))                AS profit,
       string_agg(DISTINCT p.brand, ', ')                    AS brands
FROM sales s
JOIN branches   b  ON b.id = s.branch_id
JOIN employees  e  ON e.id = s.employee_id
JOIN sale_items si ON si.sale_id = s.id
JOIN products   p  ON p.id = si.product_id
WHERE s.sold_at >= '2026-10-15' AND s.sold_at < '2026-10-20'
GROUP BY b.name, e.name
ORDER BY profit DESC;
ফলাফল
